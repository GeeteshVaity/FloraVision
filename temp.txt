class LoadingScreen extends ConsumerStatefulWidget {
  const LoadingScreen({super.key, this.imageBytes});

  final Uint8List? imageBytes;

  @override
  ConsumerState<LoadingScreen> createState() => _LoadingScreenState();
}

class _LoadingScreenState extends ConsumerState<LoadingScreen> {
  bool _hasStarted = false;

  @override
  void initState() {
    super.initState();

    WidgetsBinding.instance.addPostFrameCallback((_) {
      _runInference();
    });
  }

  Future<void> _runInference() async {
    if (_hasStarted) {
      return;
    }

    _hasStarted = true;

    final imageBytes = widget.imageBytes;

    if (imageBytes == null) {
      if (!mounted) {
        return;
      }

      context.pop();
      return;
    }

    try {
      final classifier = ref.read(flowerClassifierProvider);
      final Prediction prediction = await classifier.predict(imageBytes);

      if (!mounted) {
        return;
      }

      context.go(
        "/result",
        extra: prediction,
      );
    } catch (error) {
      if (!mounted) {
        return;
      }

      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text('Prediction failed: $error'),
        ),
      );

      context.pop();
    }
  }

  @override
  Widget build(BuildContext context) {

    return Scaffold(

      body: SafeArea(

        child: Center(

          child: Column(

            mainAxisAlignment: MainAxisAlignment.center,

            children: [

              const Icon(
                Icons.search,
                size: 90,
              ),

              const SizedBox(height: 30),

              const Text(
                "Analyzing Plant...",

                ),
              ),

              const SizedBox(height: 40),

              const CircularProgressIndicator(),

            ],
          ),
        ),
      ),
    );
  }
}

import 'dart:math' as math;
import 'package:flutter/foundation.dart';
import 'package:flutter/services.dart';
import 'package:image/image.dart' as img;
import 'package:tflite_flutter/tflite_flutter.dart';

import '../models/prediction.dart';

class FlowerClassifier {
  Interpreter? _interpreter;
  List<String> _labels = [];

  bool get isLoaded => _interpreter != null;

  void _ensureLoaded() {
    if (_interpreter == null) {
      throw StateError('TensorFlow Lite model is not loaded yet.');
    }
  }

  Future<void> loadModel() async {
    if (_interpreter != null) {
      return;
    }

    try {
      _interpreter = await Interpreter.fromAsset(
        'assets/models/flower_model.tflite',
      );

      final labelData =
          await rootBundle.loadString('assets/labels/labels.txt');

      _labels = labelData
          .split('\n')
          .map((e) => e.trim())
          .where((e) => e.isNotEmpty)
          .toList();

      debugPrint("================================");
      debugPrint("Model Loaded Successfully");
      debugPrint("Labels Loaded: ${_labels.length}");
      debugPrint(_labels.toString());
      debugPrint(
        'Input tensor: ${_interpreter!.getInputTensor(0).shape} '
        '${_interpreter!.getInputTensor(0).type}',
      );
      debugPrint(
        'Output tensor: ${_interpreter!.getOutputTensor(0).shape} '
        '${_interpreter!.getOutputTensor(0).type}',
      );
      debugPrint("================================");
    } catch (e) {
      debugPrint("Error Loading Model");
      debugPrint(e.toString());
    }
  }

  Future<Prediction> predict(Uint8List imageBytes) async {
    _ensureLoaded();

    final interpreter = _interpreter!;
    final decodedImage = img.decodeImage(imageBytes);

    if (decodedImage == null) {
      throw StateError('Unable to decode the selected image.');
    }

    final inputTensor = interpreter.getInputTensor(0);
    final inputShape = inputTensor.shape;

    if (inputShape.length < 4) {
      throw StateError('Unsupported model input shape: $inputShape');
    }

    final height = inputShape[1];
    final width = inputShape[2];

    final resizedImage = img.copyResize(
      img.bakeOrientation(decodedImage),
      width: width,
      height: height,
    );

    final input = _buildInputTensor(resizedImage, inputTensor);
    final outputTensor = interpreter.getOutputTensor(0);
    final output = _createOutputBuffer(outputTensor);

    interpreter.run(input, output);

    final scores = _readScores(output, outputTensor);
    final probabilities = _softmax(scores);
    final bestIndex = _bestScoreIndex(probabilities);
    final label = bestIndex < _labels.length
        ? _labels[bestIndex]
        : 'Class ${bestIndex + 1}';

    return Prediction(
      label: label,
      confidence: probabilities[bestIndex],
    );
  }

  Object _buildInputTensor(img.Image image, Tensor tensor) {
    final inputShape = tensor.shape;
    final height = inputShape[1];
    final width = inputShape[2];
    final inputType = tensor.type;

    if (inputType == TensorType.float32) {
      return List.generate(1, (_) {
        return List.generate(height, (y) {
          return List.generate(width, (x) {
            final pixel = image.getPixel(x, y);
            return <double>[
              pixel.r / 255.0,
              pixel.g / 255.0,
              pixel.b / 255.0,
            ];
          });
        });
      });
    }

    if (inputType == TensorType.uint8 || inputType == TensorType.int8) {
      final params = tensor.params;
      return List.generate(1, (_) {
        return List.generate(height, (y) {
          return List.generate(width, (x) {
            final pixel = image.getPixel(x, y);
            return <int>[
              _quantize(pixel.r / 255.0, params, inputType),
              _quantize(pixel.g / 255.0, params, inputType),
              _quantize(pixel.b / 255.0, params, inputType),
            ];
          });
        });
      });
    }

    throw StateError('Unsupported input tensor type: ${inputType.name}');
  }

  Object _createOutputBuffer(Tensor tensor) {
    final classCount = tensor.shape.isNotEmpty ? tensor.shape.last : _labels.length;

    if (tensor.type == TensorType.float32) {
      return List.generate(1, (_) => List<double>.filled(classCount, 0.0));
    }

    if (tensor.type == TensorType.uint8 || tensor.type == TensorType.int8) {
      return List.generate(1, (_) => List<int>.filled(classCount, 0));
    }

    throw StateError('Unsupported output tensor type: ${tensor.type.name}');
  }

  List<double> _readScores(Object output, Tensor tensor) {
    final rawScores = (output as List).first as List;

    if (tensor.type == TensorType.float32) {
      return rawScores.map((value) => (value as num).toDouble()).toList();
    }

    final params = tensor.params;
    return rawScores
        .map((value) => _dequantize(value as num, params))
        .toList();
  }

  int _bestScoreIndex(List<double> scores) {
    var bestIndex = 0;
    var bestScore = scores.first;

    for (var i = 1; i < scores.length; i++) {
      if (scores[i] > bestScore) {
        bestScore = scores[i];
        bestIndex = i;
      }
    }

    return bestIndex;
  }

  int _quantize(double value, QuantizationParams params, TensorType tensorType) {
    if (params.scale == 0) {
      return value.round();
    }

    final quantized = (value / params.scale + params.zeroPoint).round();

    if (tensorType == TensorType.int8) {
      return quantized.clamp(-128, 127);
    }

    return quantized.clamp(0, 255);
  }

  double _dequantize(num value, QuantizationParams params) {
    return (value.toDouble() - params.zeroPoint) * params.scale;
  }

  List<double> _softmax(List<double> values) {
    if (values.isEmpty) {
      return values;
    }

    final maxValue = values.reduce(math.max);
    final expValues = values
        .map((value) => math.exp(value - maxValue))
        .toList(growable: false);
    final sum = expValues.fold<double>(0.0, (total, value) => total + value);

    if (sum == 0) {
      return List<double>.filled(values.length, 0.0);
    }

    return expValues.map((value) => value / sum).toList(growable: false);
  }

  List<String> get labels => _labels;

  Interpreter get interpreter => _interpreter!;
}
